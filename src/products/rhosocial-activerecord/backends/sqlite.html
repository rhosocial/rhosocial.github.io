SQLite

内置、零依赖、单文件数据库。同步/异步双模式，PRAGMA 精细控制。

Features

SQLite 特性。

01 / Zero Config

零配置

开箱即用，无需安装服务器。默认启用 WAL 模式与外键约束。

02 / Single File

单文件

整个数据库是一个文件，便于分发。也支持 :memory: 内存数据库。

03 / ACID

完整 ACID

事务保证原子性、一致性、隔离性、持久性。DEFERRED / IMMEDIATE / EXCLUSIVE 三种 BEGIN 模式。

04 / Extensions

扩展生态

FTS5 全文搜索、JSON1、R-Tree、Geopoly 等虚拟表扩展。扩展注册表运行时检测可用性。

Connection

连接配置。

SQLiteConnectionConfig 继承 ConnectionConfig，混入 PRAGMA / Driver / Storage 三组配置。

connection_config.py
from rhosocial.activerecord.backend.impl.sqlite import SQLiteBackend
from rhosocial.activerecord.backend.impl.sqlite.config import (
    SQLiteConnectionConfig,
    SQLiteInMemoryConfig,
    SQLiteTempFileConfig,
)

# 文件数据库（自动启用默认 PRAGMA）
config = SQLiteConnectionConfig(database='app.db')
backend = SQLiteBackend(connection_config=config)
backend.connect()

# 内存数据库（自动优化 PRAGMA）
mem_config = SQLiteInMemoryConfig()
mem_backend = SQLiteBackend(connection_config=mem_config)

# 临时文件数据库（关闭时自动删除）
temp_config = SQLiteTempFileConfig()
temp_backend = SQLiteBackend(connection_config=temp_config)

# 快捷方式：直接传 database 参数
backend = SQLiteBackend(database='app.db')

Pragmas

默认 PRAGMA。

SQLiteConnectionConfig 自动启用以下 PRAGMA，兼顾安全性与性能：

default_pragmas.py
DEFAULT_PRAGMAS = {
    "foreign_keys":     "ON",       # 启用外键约束
    "journal_mode":     "WAL",       # 写前日志模式，允许并发读
    "synchronous":      "FULL",      # 完全同步，最安全
    "wal_autocheckpoint": "1000",   # WAL 自动检查点阈值
    "wal_checkpoint":   "FULL",      # 检查点模式
}

# 自定义 PRAGMA
config = SQLiteConnectionConfig(
    database='app.db',
    pragmas={
        "foreign_keys": "ON",
        "journal_mode": "WAL",
        "cache_size": "-64000",  # 64MB 页缓存
        "mmap_size": "268435456", # 256MB 内存映射
    }
)

# 内存数据库自动覆盖 PRAGMA
# journal_mode → MEMORY, synchronous → OFF, temp_store → MEMORY

Dialect

SQLiteDialect 版本感知。

SQLiteDialect 在初始化时接受版本元组，动态启用/禁用协议方法。

dialect_version.py
from rhosocial.activerecord.backend.impl.sqlite.dialect import SQLiteDialect

# 默认版本 3.35.0（启用 RETURNING）
dialect = SQLiteDialect()

# 指定旧版本 → 功能降级
old_dialect = SQLiteDialect(version=(3, 24, 0))
old_dialect.supports_returning_clause()  # False
old_dialect.supports_window_functions()  # False
old_dialect.supports_upsert()            # True (3.24.0+)

# 版本对应功能表
# 3.8.3  — CTE / Recursive CTE
# 3.25.0 — Window Functions, RENAME COLUMN
# 3.31.0 — Generated Columns (STORED/VIRTUAL)
# 3.35.0 — RETURNING, DROP COLUMN, MATERIALIZED CTE
# 3.38.0 — JSON1 内置（不再依赖扩展）

INSERT Syntax

INSERT 语法扩展。

insert_syntax.py
from rhosocial.activerecord.backend.expression import InsertExpression, ValuesSource, Literal

# 标准 INSERT ... ON CONFLICT (3.24.0+)
insert = InsertExpression(dialect=dialect, into='users',
    columns=['id', 'name'],
    source=ValuesSource(dialect, [[Literal(dialect, 1), Literal(dialect, 'Alice')]]),
    on_conflict=...)

# SQLite 专有：INSERT OR REPLACE（删除旧行，插入新行）
insert_or_replace = InsertExpression(
    dialect=dialect, into='users',
    columns=['id', 'name'],
    source=ValuesSource(dialect, [[Literal(dialect, 1), Literal(dialect, 'Alice')]]),
    dialect_options={'or_replace': True}
)

# SQLite 专有：INSERT OR IGNORE（静默跳过冲突行）
insert_or_ignore = InsertExpression(
    dialect=dialect, into='users',
    columns=['id', 'name'],
    source=ValuesSource(dialect, [[Literal(dialect, 1), Literal(dialect, 'Alice')]]),
    dialect_options={'or_ignore': True}
)

FTS5 Example

FTS5 全文搜索示例。

fts5_example.py
from rhosocial.activerecord.backend.impl.sqlite import SQLiteBackend

backend = SQLiteBackend(database='docs.db')
backend.connect()
dialect = backend.dialect

# 创建 FTS5 虚拟表
sql, params = dialect.format_fts5_create_virtual_table(
    table_name='articles',
    columns=['title', 'body'],
    tokenizer='unicode61',
)
# → CREATE VIRTUAL TABLE "articles" USING fts5(title, body, tokenize=unicode61)

# 全文搜索 MATCH
sql, params = dialect.format_fts5_match_expression(
    table_name='articles', query='python async'
)
# → ("articles" MATCH ?, ('python async',))

# BM25 排名
sql, params = dialect.format_fts5_rank_expression(
    table_name='articles', weights=[10.0, 1.0]
)
# → bm25("articles", 10.0, 1.0)

# 高亮搜索结果
sql, params = dialect.format_fts5_highlight_expression(
    table_name='articles', column='body', query='python',
    prefix_marker='<mark>', suffix_marker='</mark>'
)

# 检测扩展可用性
if dialect.is_extension_available('fts5'):
    features = dialect.get_supported_extension_features('fts5')

Pragma API

PRAGMA 操作示例。

pragma_api.py
from rhosocial.activerecord.backend.impl.sqlite.pragma import (
    PragmaCategory, get_pragma_info, get_pragmas_by_category
)

# 查询单个 PRAGMA 元数据
info = get_pragma_info('journal_mode')
info.get_sql()        # → "PRAGMA journal_mode"
info.get_set_sql('WAL') # → "PRAGMA journal_mode = WAL"

# 按类别浏览 PRAGMA
perf_pragmas = get_pragmas_by_category(PragmaCategory.PERFORMANCE)

# 通过方言运行时查询
dialect = backend.dialect
sql = dialect.get_pragma_sql('cache_size')
# → "PRAGMA cache_size"

sql = dialect.set_pragma_sql('cache_size', '-64000')
# → "PRAGMA cache_size = -64000"

# 检测 PRAGMA 可用性
dialect.is_pragma_available('journal_mode')  # True
Transaction

事务控制。

SQLiteTransactionManager 支持三种 BEGIN 模式，默认 SERIALIZABLE 隔离级别。

transaction_control.py
from rhosocial.activerecord.backend.transaction import IsolationLevel

# 标准事务（默认 BEGIN IMMEDIATE TRANSACTION）
with backend.transaction():
    backend.execute("INSERT INTO users ...", ())

# 指定隔离级别
with backend.transaction(isolation_level=IsolationLevel.SERIALIZABLE):
    ...

# SQLite 仅支持两种隔离级别
# SERIALIZABLE      → BEGIN IMMEDIATE (默认)
# READ_UNCOMMITTED  → BEGIN DEFERRED + PRAGMA read_uncommitted=1

# 直接控制 BEGIN 类型
tm = backend.transaction_manager
tm.begin_type = 'EXCLUSIVE'  # BEGIN EXCLUSIVE TRANSACTION
# DEFERRED   — 不获取锁，首次读写时加锁
# IMMEDIATE  — 立即获取 RESERVED 锁（推荐写事务）
# EXCLUSIVE  — 立即获取 EXCLUSIVE 锁（独占访问）

# Savepoint 支持
with backend.transaction():
    backend.execute("INSERT ...", ())
    with backend.transaction():  # SAVEPOINT
        backend.execute("UPDATE ...", ())
        # ROLLBACK TO SAVEPOINT → 自动回滚

Async

异步支持。

AsyncSQLiteBackend / AsyncSQLiteTransactionManager 懒加载，需安装 aiosqlite。

async_sqlite.py
from rhosocial.activerecord.backend.impl.sqlite import AsyncSQLiteBackend
# 需安装: pip install rhosocial-activerecord[async]

backend = AsyncSQLiteBackend(database='app.db')
await backend.connect()

async with backend.transaction():
    result = await backend.execute("SELECT ...", ())

await backend.disconnect()

# 懒加载机制：导入同步组件不需要 aiosqlite
from rhosocial.activerecord.backend.impl.sqlite import SQLiteBackend  # ✓ 无需 aiosqlite
from rhosocial.activerecord.backend.impl.sqlite import AsyncSQLiteBackend  # ✗ 需要 aiosqlite
Environment

env_config.py
from rhosocial.activerecord.backend.impl.sqlite.config import SQLiteConnectionConfig

# 从环境变量创建配置（前缀 SQLITE_）
config = SQLiteConnectionConfig.from_env(prefix="SQLITE_")

# 支持的环境变量：
# SQLITE_DATABASE         — 数据库文件路径
# SQLITE_URI              — 启用 URI 文件名 (true/false)
# SQLITE_TIMEOUT          — 连接超时（秒）
# SQLITE_ISOLATION_LEVEL  — 隔离级别
# SQLITE_DETECT_TYPES     — 类型检测标志
# SQLITE_CACHED_STATEMENTS — 缓存语句数
# SQLITE_AUTOCOMMIT       — 自动提交模式
# SQLITE_DELETE_ON_CLOSE  — 关闭时删除文件
# SQLITE_PRAGMA_*         — 自定义 PRAGMA（如 SQLITE_PRAGMA_CACHE_SIZE）
