Set Operation

UNION / INTERSECT / EXCEPT 查询集合操作。

UNION

并集。

main.py
from rhosocial.activerecord.query import SetOperationQuery

# UNION (去重)
results = (SetOperationQuery()
    .union(User.query().where(User.c.role == 'admin'))
    .union(User.query().where(User.c.role == 'moderator'))
    .all())

# UNION ALL (保留重复) — 使用 all_=True
results = (SetOperationQuery()
    .union(User.query().where(User.c.role == 'admin'), all_=True)
    .union(User.query().where(User.c.role == 'moderator'), all_=True)
    .all())
INTERSECT

交集。

main.py
# 交集 — 同时出现在两个查询中的记录
results = (SetOperationQuery()
    .intersect(User.query().where(User.c.is_active == True))
    .intersect(User.query().where(User.c.is_verified == True))
    .all())
EXCEPT

差集。

main.py
# 差集 — 出现在第一个查询但不在第二个查询中的记录
results = (SetOperationQuery()
    .except_(User.query().where(User.c.role == 'banned'))
    .all())
