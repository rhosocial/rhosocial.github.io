CTE

公共表表达式，化繁为简。

API

CTEQuery 接口。

main.py
from rhosocial.activerecord.query import CTEQuery, AsyncCTEQuery

# 非递归 CTE
cte = (CTEQuery()
    .with_cte('active_users', User.query().where(User.c.is_active == True))
    .select_from('active_users')
    .all())

# 递归 CTE (树形结构)
cte = (CTEQuery()
    .with_recursive_cte('org_tree',
        base_query=Department.query().where(Department.c.parent_id.is_null()),
        recursive_query=Department.query().join('org_tree', Department.c.parent_id == Department.c.id))
    .select_from('org_tree')
    .all())
