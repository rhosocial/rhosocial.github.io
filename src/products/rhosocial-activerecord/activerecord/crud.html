← ActiveRecord

CRUD

Create, Read, Update, Delete。

Create

创建记录。

main.py
# 方式一：实例化后保存
user = User(username='Alice', email='alice@example.com')
user.save()

# 方式二：直接创建并保存
user = User.create(username='Bob', email='bob@example.com')
Read

读取记录。

main.py
# 按主键获取 (find_one 接受 PK 值作为参数)
user = User.find_one(1)

# 条件查询 (query().where().one())
user = User.query().where(User.c.username == 'Alice').one()

# 获取全部 (query().all())
users = User.query().all()
Update

更新记录。

main.py
# 修改属性后保存
user = User.find_one(1)
user.email = 'new@example.com'
user.save()

# 批量更新 (query().where().update())
User.query().where(User.c.username == 'Alice').update({email: 'alice@new.com'})
Delete

删除记录。

main.py
# 实例方法删除
user = User.find_one(1)
user.delete()

# 条件删除 (query().where().delete())
User.query().where(User.c.username == 'Alice').delete()
